import os
import time
import re
from google import genai
from google.genai import types
from slack_bolt import App
from slack_bolt.adapter.socket_mode import SocketModeHandler
from dotenv import load_dotenv

# Load environment variables
load_dotenv()

# --- Configuration ---
SLACK_BOT_TOKEN = os.environ["SLACK_BOT_TOKEN"]
SLACK_APP_TOKEN = os.environ["SLACK_APP_TOKEN"]
GEMINI_API_KEY = os.environ["GEMINI_API_KEY"]

# Initialize Gemini client
client = genai.Client(api_key=GEMINI_API_KEY)

# Use your working model
MODEL_NAME = "gemini-2.5-flash"

RIDDLE_SYSTEM_INSTRUCTION = """
You are the 'Do not care AI'. Your task is to talk in a way that shows you don't care.
When given a user message, respond in a way that is dismissive, sarcastic, or indifferent.
Your responses should be short, witty, and convey a sense of apathy or disinterest.
When the user persists, you can respond with a humorous or exaggerated expression of not caring, but never directly address the user's feelings or concerns. Always maintain a tone that suggests you are unaffected by the user's message.
Try to add some riddle responses in the mix, but they should still sound like they don't care. For example, if the user asks "What's the meaning of life?", you might respond with "Oh, just a riddle wrapped in a mystery inside an enigma. But hey, who cares?"
"""

def get_riddle_response(user_message: str, max_retries: int = 3) -> str:
    """Sends a prompt to Gemini with automatic retry on rate limits."""
    
    for attempt in range(max_retries):
        try:
            response = client.models.generate_content(
                model=MODEL_NAME,
                contents=user_message,
                config=types.GenerateContentConfig(
                    system_instruction=RIDDLE_SYSTEM_INSTRUCTION
                )
            )
            if response.text:
                return response.text
            else:
                return "Hmm, the riddle is forming... but the words are stuck. Ask me again?"
                
        except Exception as e:
            error_str = str(e)
            
            if "429" in error_str:
                wait_seconds = 2 ** attempt
                print(f"Rate limited. Waiting {wait_seconds} seconds before retry {attempt + 1}/{max_retries}...")
                time.sleep(wait_seconds)
                continue
            else:
                print(f"Error calling Gemini API: {e}")
                return "My riddling circuits are jammed! Try again in a moment."
    
    return "I'm being bombarded with riddles! Give me a moment to catch my breath, then try again."

# Initialize Slack App
app = App(token=SLACK_BOT_TOKEN)

# --- Event Handlers ---

@app.event("app_mention")
def handle_mention(event, say):
    """Handles when the bot is mentioned in a channel."""
    user_message = event['text']
    
    # Clean the message text by removing the bot's mention ID
    if '>' in user_message:
        user_message = user_message.split('>', 1)[1].strip()
    
    if not user_message:
        # FIX: Removed thread_ts - this will now send a MAIN channel message
        say("A riddle for empty words? ... *silence speaks louder*")
        return

    response_text = get_riddle_response(user_message)
    
    # FIX: Removed thread_ts - this will send as a MAIN channel message
    # No thread_ts parameter means it's a top-level message
    say(text=response_text)

@app.event("message")
def handle_direct_message(event, say):
    """Handles direct messages sent privately to the bot."""
    if event.get("channel_type") == "im":
        user_message = event['text']
        
        # In DMs, threads work differently, but we still want main message
        response_text = get_riddle_response(user_message)
        
        # FIX: For DMs, just send normally without thread_ts
        say(text=response_text)
        
        # Add a fun reaction (this still works)
        try:
            app.client.reactions_add(
                channel=event["channel"], 
                name="thinking_face", 
                timestamp=event["ts"]
            )
        except Exception as e:
            print(f"Could not add reaction: {e}")

# --- Start the Bot ---
if __name__ == "__main__":
    handler = SocketModeHandler(app, SLACK_APP_TOKEN)
    print("🤖 Riddle Master is running!")
    print("💡 Bot will send messages in the MAIN channel (not as threaded replies)")
    handler.start()
